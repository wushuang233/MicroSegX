From outside the cluster, the MicroSegX URL is:
http://{{ .Values.manager.ingress.host }}
{{- else if and .Values.manager.enabled .Values.manager.ingress.enabled .Values.manager.ingress.tls}}
From outside the cluster, the MicroSegX URL is:
https://{{ .Values.manager.ingress.host }}
{{- else if not .Values.openshift }}
Get the MicroSegX URL by running these commands:
{{- if contains "NodePort" .Values.manager.svc.type }}
  NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services microsegx-service-webui)
  NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo https://$NODE_IP:$NODE_PORT
{{- else if contains "ClusterIP" .Values.manager.svc.type }}
  CLUSTER_IP=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.clusterIP}" services microsegx-service-webui)
  TARGET_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].targetPort}" services microsegx-service-webui)
  echo https://$CLUSTER_IP:$TARGET_PORT
{{- else if contains "LoadBalancer" .Values.manager.svc.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        Watch the status by running 'kubectl get svc --namespace {{ .Release.Namespace }} -w microsegx-service-webui'

  SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} microsegx-service-webui -o jsonpath="{.status.loadBalancer.ingress[0].ip}")
  TARGET_PORT=$(kubectl get svc --namespace {{ .Release.Namespace }} microsegx-service-webui -o jsonpath="{.spec.ports[0].targetPort}")
  echo https://$SERVICE_IP:$TARGET_PORT
{{- end }}
{{- end }}
